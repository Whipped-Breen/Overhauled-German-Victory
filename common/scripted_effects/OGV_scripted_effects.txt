siberian_nationalist_independence = {
	set_temp_variable = { siberian_nationalist_country = this }
	division_template = {
		name = "Independence Division"
		is_locked = yes
		priority = 1
		template_counter = 3
		regiments = {
			infantry = { x = 0 y = 0 }
			infantry = { x = 0 y = 1 }
			infantry = { x = 1 y = 0 }
			infantry = { x = 1 y = 1 }
		}
	}
	every_core_state = {
		limit = {
			is_owned_and_controlled_by = SOV
			NOT = { state = 406 }
			NOT = { state = 407 }
#			NOT = { state = 657 }
#			NOT = { state = 409 }
#			NOT = { state = 408 }
		}
		var:siberian_nationalist_country = { transfer_state = PREV }
		if = {
			limit = {
				NOT = { is_owned_and_controlled_by = RSB }
			}
			SOV = {
				delete_unit = { 
					state = PREV 
					disband = no 
				}
			}
			if = {
				limit = { owner = { not = { tag = FER } } JAP = { is_subject = yes } }
				create_unit = { division = "name = \"Independence Division\" division_template = \"Independence Division\" start_experience_factor = 0.05" owner = var:siberian_nationalist_country count = 1 }
			}
		}
	}
	random_owned_state = {
		limit = { is_capital = yes }
		create_unit = { 
			division = "name = \"Independence Division\" division_template = \"Independence Division\" start_experience_factor = 0.05 start_equipment_factor = 0.95" owner = var:siberian_nationalist_country count = 1
		}
	}
	# Renaming of cities where apropriate 
	if = {
		limit = { var:siberian_nationalist_country = { tag = KYR }}
		set_province_name = {
			id = 12583
			name = SOV_Bishkek
		}
	}
}
